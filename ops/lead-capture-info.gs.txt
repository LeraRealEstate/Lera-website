// Lera lead capture (runs as info@lera-realestate.com)
// Landing pages post each lead here; it lands as a new row in the Google Sheet.
// Facebook / Instagram ad leads go to "Facebook leads"; everything else to "Website leads".

var SHEET_ID = '1_uUz0BLJT80qdkv2A_wfQEVqvDkeH4QXMyo2CBVooq4';
var FB_TAB = 'Facebook leads';
var WEB_TAB = 'Website leads';
var LEAD_KEY = 'lera-7Qm4xV9pT2'; // must match the key sent by the landing page

function doPost(e) {
  var p = (e && e.parameter) || {};
  if (p._honey) return out_({ ok: true });
  if (p.key !== LEAD_KEY) return out_({ ok: false });
  if (p.action === 'book') return out_(book_(p));

  var src = String(p.utm_source || p.source || '').toLowerCase();
  var isAd = /facebook|instagram|^fb|^ig|meta/.test(src) || !!p.fbclid;
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var sh = ss.getSheetByName(isAd ? FB_TAB : WEB_TAB) || ss.getSheetByName(FB_TAB);

  var name = String(p.name || '').trim();
  var first = p.first_name || name.split(/\s+/)[0] || '';
  var last = p.last_name || name.split(/\s+/).slice(1).join(' ');
  var when = Utilities.formatDate(new Date(), 'Europe/London', 'dd/MM/yyyy HH:mm');
  var action = p.action === 'call' ? 'Booked / requested a call' : 'Downloaded HMO guide';
  var note = when + ' - ' + action +
    (p.utm_campaign ? ' (campaign: ' + p.utm_campaign + ')' : '') +
    (p.utm_content ? ' (ad: ' + p.utm_content + ')' : '');

  var values = {
    'where from': isAd ? (src.indexOf('insta') > -1 ? 'Instagram' : 'Facebook') : 'Website',
    'first name': first,
    'last name': last,
    'phone number': p.phone ? "'" + p.phone : '',
    'phone': p.phone ? "'" + p.phone : '',
    'email': p.email || '',
    'email address': p.email || '',
    'amount of money': p.budget || '',
    'notes': note,
    'status': 'New',
    'campaign': p.utm_campaign || '',
    'ad': p.utm_content || '',
    'landing page': p.page || ''
  };

  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    ensureCols_(sh);
    var lastCol = sh.getLastColumn();
    var headers = sh.getRange(1, 1, 1, lastCol).getValues()[0];
    var row = [], used = {};
    for (var i = 0; i < headers.length; i++) {
      var h = String(headers[i]).trim().toLowerCase().replace(/:$/, '').trim();
      var v = '';
      if (values.hasOwnProperty(h) && !used[h]) { v = values[h]; used[h] = true; }
      row.push(v);
    }
    sh.appendRow(row);
  } finally {
    lock.releaseLock();
  }
  try { notifyLead_(p); } catch (err) { console.error(err); }
  return out_({ ok: true });
}

function doGet(e) {
  var a = e && e.parameter && e.parameter.action;
  if (a === 'slots') return out_(slotsResult_());
  return out_({ ok: true, service: 'Lera lead capture' });
}

function out_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

function ensureCols_(sh) {
  var h = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(function (x) { return String(x).trim().toLowerCase(); });
  ['Campaign', 'Ad', 'Landing page'].forEach(function (c) {
    if (h.indexOf(c.toLowerCase()) === -1) { sh.getRange(1, sh.getLastColumn() + 1).setValue(c); h.push(c.toLowerCase()); }
  });
}

// Runs straight after a lead is saved
function notifyLead_(p) {
  var first = cap_(String(p.name || '').trim().split(/\s+/)[0] || 'there');
  var email = String(p.email || '').trim();
  if (p.action === 'call') {
    return; // they pick a time on the page; booking sends the confirmation
    if (email) sendMail_(email, 'Your call with Lera: pick a time',
      'Hi ' + first + ',\n\nThanks for asking for a call. If you haven\'t picked a time yet, choose one here:\n' + BOOK_URL +

    sendWa_(p.phone, 'lera_call_request', [first]);

      'Hi ' + first + ',\n\nHere\'s your copy of the Lera HMO guide: ' + GUIDE_URL + '\n\nWant to talk it through? Book a free call with Akiva: ' + BOOK_URL + '\n\nLera Real Estate', guideEmail_(first));

}

// Runs every minute: reminds people 1 hour and 5 minutes before their call
function callReminders() {
  var cal = CalendarApp.getCalendarById(CAL_ID) || CalendarApp.getDefaultCalendar();
  var me = Session.getEffectiveUser().getEmail().toLowerCase();
  var now = new Date();
  var events = cal.getEvents(now, new Date(now.getTime() + 66 * 60000));
  events.forEach(function (ev) {
    var guests = ev.getGuestList().filter(function (g) { return g.getEmail().toLowerCase() !== me; });
    if (!guests.length) return;                       // not a booked call
    var g = guests[0], email = g.getEmail();
    var mins = (ev.getStartTime().getTime() - now.getTime()) / 60000;
    var lead = findLead_(email);
    var first = cap_((lead.first || g.getName() || '').split(/\s+/)[0] || 'there');
    var phone = lead.phone || phoneFrom_(ev.getDescription());
    var time = Utilities.formatDate(ev.getStartTime(), 'Europe/London', 'h:mm a');
    if (!/^Free (video |phone )?call with Lera/.test(ev.getTitle())) return; // only Lera calls
    var link = meetLink_(ev);
    if (mins <= 62 && mins > 50 && !ev.getTag('lera_r60')) {
      sendMail_(email, 'In one hour: your call with Akiva at ' + time,
        'Hi ' + first + ', your free call with Akiva is in one hour, at ' + time + '.' + (link ? ' Join: ' + link : ' Akiva will call you on ' + phone + '.'), reminderEmail_(first, time, !!link, link, phone, false));
      ev.setTag('lera_r60', '1');
    }
    if (mins <= 6 && mins > 0 && !ev.getTag('lera_r5')) {
      sendMail_(email, 'Starting now: your call with Akiva',
        'Hi ' + first + ', your call with Akiva starts in 5 minutes, at ' + time + '.' + (link ? ' Join: ' + link : ' Akiva will call you on ' + phone + '.'), reminderEmail_(first, time, !!link, link, phone, true));
      sendWa_(phone, 'lera_call_reminder_5min', [first, time]);
      ev.setTag('lera_r5', '1');
    }
  });
}

function findLead_(email) {
  var ss = SpreadsheetApp.openById(SHEET_ID), out = {};
  [FB_TAB, WEB_TAB].forEach(function (t) {
    var sh = ss.getSheetByName(t); if (!sh || out.phone) return;
    var data = sh.getDataRange().getValues(), h = data[0].map(function (x) { return String(x).trim().toLowerCase().replace(/:$/, ''); });
    var ce = h.indexOf('email') > -1 ? h.indexOf('email') : h.indexOf('email address');
    var cp = h.indexOf('phone number') > -1 ? h.indexOf('phone number') : h.indexOf('phone');
    var cf = h.indexOf('first name');
    for (var r = data.length - 1; r > 0; r--) {
      if (String(data[r][ce]).trim().toLowerCase() === email.toLowerCase()) {
        out.phone = cp > -1 ? String(data[r][cp]).replace(/^'/, '') : ''; out.first = cf > -1 ? String(data[r][cf]) : ''; break;
      }
    }
  });
  return out;
}

function phoneFrom_(text) { var m = String(text || '').match(/(\+?44|0)\s?7[\d\s]{9,12}/); return m ? m[0] : ''; }
function meetLink_(ev) { var m = String(ev.getDescription() || '').match(/https:\/\/meet\.google\.com\/[a-z\-]+/); return m ? m[0] : ''; }

function sendMail_(to, subject, body, html) {
  // Only ever sends from info@lera-realestate.com
  try {
    if (Session.getEffectiveUser().getEmail().toLowerCase() !== REPLY_TO) { console.error('Not running as info@ - email NOT sent to ' + to); return; }
    var o = { name: FROM_NAME, replyTo: REPLY_TO }; if (html) o.htmlBody = html;
    GmailApp.sendEmail(to, subject, body, o);
  }
  catch (e) { console.error('email failed ' + to + ': ' + e); }
}

// WhatsApp needs approved message templates with these names (body variables {{1}}, {{2}}).
function sendWa_(phone, template, vars) {
  var props = PropertiesService.getScriptProperties();
  var tok = props.getProperty('WA_TOKEN') || WA_TOKEN, pid = props.getProperty('WA_PHONE_ID') || WA_PHONE_ID;
  if (!tok || !pid || !phone) return;
  var n = String(phone).replace(/[^\d+]/g, '');
  if (/^07\d{9}$/.test(n)) n = '44' + n.slice(1);
  n = n.replace(/^\+/, '').replace(/^0044/, '44');
  var payload = { messaging_product: 'whatsapp', to: n, type: 'template',
    template: { name: template, language: { code: 'en_GB' },
      components: [{ type: 'body', parameters: vars.map(function (v) { return { type: 'text', text: String(v) }; }) }] } };
  try {
    var r = UrlFetchApp.fetch('https://graph.facebook.com/v21.0/' + pid + '/messages', {
      method: 'post', contentType: 'application/json', muteHttpExceptions: true,
      headers: { Authorization: 'Bearer ' + tok }, payload: JSON.stringify(payload) });
    if (r.getResponseCode() >= 300) console.error('WhatsApp ' + template + ' failed: ' + r.getContentText());
  } catch (e) { console.error('WhatsApp failed: ' + e); }
}

// Run once: adds Campaign / Ad / Landing page columns and starts the reminder clock

  [FB_TAB, WEB_TAB].forEach(function (t) {

    var h = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(function (x) { return String(x).trim().toLowerCase(); });
    ['Campaign', 'Ad', 'Landing page'].forEach(function (c) {

    });
  });
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'callReminders') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('callReminders').timeBased().everyMinutes(1).create();
  var cal = CalendarApp.getCalendarById(CAL_ID) || CalendarApp.getDefaultCalendar();
  console.log('Reminders running on calendar: ' + cal.getName() + '. Upcoming booked calls: ' +
    cal.getEvents(new Date(), new Date(Date.now() + 14 * 864e5)).filter(function (e) { return e.getGuestList().length; }).length);
}

// ===================== BOOKING (calls booked straight from the landing page) =====================
var SLOT_DAYS = 10;           // working days shown
// England & Wales bank holidays (gov.uk). No calls are offered on these days, or at weekends.
var BANK_HOLIDAYS = ['2026-01-01','2026-04-03','2026-04-06','2026-05-04','2026-05-25','2026-08-31','2026-12-25','2026-12-28',
  '2027-01-01','2027-03-26','2027-03-29','2027-05-03','2027-05-31','2027-08-30','2027-12-27','2027-12-28',
  '2028-01-03','2028-04-14','2028-04-17','2028-05-01','2028-05-29','2028-08-28','2028-12-25','2028-12-26'];
function closedDay_(d) { var tz = 'Europe/London'; return +Utilities.formatDate(d, tz, 'u') > 5 || BANK_HOLIDAYS.indexOf(Utilities.formatDate(d, tz, 'yyyy-MM-dd')) > -1; }
var SLOT_FIRST_HOUR = 9;      // first call 9:00
var SLOT_LAST_START = '16:30';// last call 16:30
var CALL_MINS = 15;           // length of the call
var MIN_NOTICE_HOURS = 2;     // earliest bookable slot from now
var OLD_SCHEDULE_TITLE = 'Free 15-minute HMO strategy call';

function cal_() { return CalendarApp.getCalendarById(CAL_ID) || CalendarApp.getDefaultCalendar(); }
function pad_(n) { return (n < 10 ? '0' : '') + n; }

function blocks_(ev) {
  if (ev.isAllDayEvent()) return false;
  try { if (ev.getTransparency && String(ev.getTransparency()) === 'TRANSPARENT') return false; } catch (x) {}
  if (ev.getTitle() === OLD_SCHEDULE_TITLE && ev.getGuestList().length === 0) return false; // the old availability block
  return true;
}

function slotsResult_() {
  var tz = 'Europe/London', now = new Date(), minStart = now.getTime() + MIN_NOTICE_HOURS * 3600000;
  var horizon = new Date(now.getTime() + 21 * 864e5);
  var evs = cal_().getEvents(now, horizon).filter(blocks_);
  var lastH = +SLOT_LAST_START.split(':')[0], lastM = +SLOT_LAST_START.split(':')[1];
  var days = [];
  for (var i = 0; i < 21 && days.length < SLOT_DAYS; i++) {
    var day = new Date(now.getTime() + i * 864e5);
    if (closedDay_(day)) continue; // weekdays only, no bank holidays
    var ymd = Utilities.formatDate(day, tz, 'yyyy-MM-dd'), times = [];
    for (var h = SLOT_FIRST_HOUR; h <= lastH; h++) {
      for (var m = 0; m < 60; m += 30) {
        if (h === lastH && m > lastM) continue;
        var s = Utilities.parseDate(ymd + ' ' + pad_(h) + ':' + pad_(m), tz, 'yyyy-MM-dd HH:mm').getTime();
        var e = s + CALL_MINS * 60000;
        if (s < minStart) continue;
        var clash = evs.some(function (ev) { return ev.getStartTime().getTime() < e && ev.getEndTime().getTime() > s; });
        if (!clash) times.push(new Date(s).toISOString());
      }
    }
    if (times.length) days.push({ date: ymd, label: Utilities.formatDate(day, tz, 'EEE d MMM'), times: times });
  }
  return { ok: true, days: days };
}

function book_(p) {
  var start = new Date(String(p.start || ''));
  if (isNaN(start.getTime()) || start.getTime() < Date.now()) return { ok: false, taken: true };
  if (closedDay_(start)) return { ok: false, taken: true };
  var end = new Date(start.getTime() + CALL_MINS * 60000);
  var name = String(p.name || '').trim(), email = String(p.email || '').trim(), phone = String(p.phone || '').trim();
  if (!name || !/@/.test(email)) return { ok: false };
  var video = p.type !== 'phone', meet = '';
  var lock = LockService.getScriptLock(); lock.waitLock(15000);
  try {
    var cal = cal_();
    var clash = cal.getEvents(start, end).filter(blocks_).length > 0;
    if (clash) return { ok: false, taken: true };
    var title = 'Free ' + (video ? 'video' : 'phone') + ' call with Lera Real Estate (Akiva): ' + name;
    var details = 'Name: ' + name + '\nPhone: ' + phone + '\nEmail: ' + email + '\nBudget: ' + (p.budget || '') +
        '\nCampaign: ' + (p.utm_campaign || '') + '\nAd: ' + (p.utm_content || '') + '\nPage: ' + (p.page || '');
    if (video) {
      var made = Calendar.Events.insert({
        summary: title,
        description: 'Free 15-minute video call with a Lera property expert. We\'ll show you some of our projects.\n\n' + details,
        start: { dateTime: start.toISOString(), timeZone: 'Europe/London' },
        end: { dateTime: end.toISOString(), timeZone: 'Europe/London' },
        attendees: [{ email: email, displayName: name }],
        reminders: { useDefault: false, overrides: [{ method: 'popup', minutes: 10 }] },
        conferenceData: { createRequest: { requestId: Utilities.getUuid(), conferenceSolutionKey: { type: 'hangoutsMeet' } } }
      }, 'primary', { conferenceDataVersion: 1, sendUpdates: 'all' });
      meet = made.hangoutLink || '';
      if (meet) Calendar.Events.patch({ description: 'Join the video call: ' + meet + '\n\nFree 15-minute video call with a Lera property expert. We\'ll show you some of our projects.\n\n' + details }, 'primary', made.id);
    } else {
      var ev = cal.createEvent(title, start, end, {
        description: 'Free 15-minute phone call with a Lera property expert. We will call you on ' + phone + '.\n\n' + details,
        guests: email, sendInvites: true });
      ev.addPopupReminder(10);
    }
  } finally { lock.releaseLock(); }
  var when = Utilities.formatDate(start, 'Europe/London', "EEEE d MMMM 'at' h:mm a");
  try { markBooked_(email, (video ? 'VIDEO' : 'PHONE') + ' call ' + Utilities.formatDate(start, 'Europe/London', 'dd/MM/yyyy HH:mm')); } catch (x) { console.error(x); }
  var first = cap_(name.split(/\s+/)[0]);
  sendMail_(email, 'You\'re booked in with Akiva: ' + when,
    'Hi ' + first + ',\n\nYou\'re booked in for a free 15-minute ' + (video ? 'video' : 'phone') + ' call with Akiva Solomon, founder of Lera Real Estate, on ' + when + '.' + (video ? '\n\nJoin here: ' + meet : '\n\nAkiva will call you on ' + phone + '.') + '\n\nLera Real Estate', bookedEmail_(first, when, video, meet, phone));
  sendWa_(phone, 'lera_call_booked', [first, when]);
  return { ok: true, when: when, meet: meet };
}

function markBooked_(email, whenTxt) {
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var tabs = [FB_TAB, WEB_TAB];
  for (var t = 0; t < tabs.length; t++) {
    var sh = ss.getSheetByName(tabs[t]); if (!sh) continue;
    var data = sh.getDataRange().getValues();
    var h = data[0].map(function (x) { return String(x).trim().toLowerCase().replace(/:$/, ''); });
    var ce = h.indexOf('email') > -1 ? h.indexOf('email') : h.indexOf('email address');
    var cn = h.indexOf('notes'), cs = h.indexOf('status');
    for (var r = data.length - 1; r > 0; r--) {
      if (String(data[r][ce]).trim().toLowerCase() === email.toLowerCase()) {
        if (cn > -1) sh.getRange(r + 1, cn + 1).setValue(String(data[r][cn]) + ' | CALL BOOKED: ' + whenTxt);
        if (cs > -1) sh.getRange(r + 1, cs + 1).setValue('Call booked');
        return;
      }
    }
  }
}

function cap_(t) { t = String(t || ''); return t.charAt(0).toUpperCase() + t.slice(1); }

/* ===================== Zoom Phone call logging =====================
   Every 15 min: reads Zoom Phone call history and notes each call to/from a lead in the Sheet.
   Script properties needed (Project Settings > Script properties): ZOOM_ACCOUNT_ID, ZOOM_CLIENT_ID, ZOOM_CLIENT_SECRET
   Then run setupZoom() once. */
function setupZoom() {
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'syncZoomCalls') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('syncZoomCalls').timeBased().everyMinutes(15).create();
  syncZoomCalls();
}

function zoomToken_() {
  var c = CacheService.getScriptCache(), t = c.get('zoom_tok'); if (t) return t;
  var p = PropertiesService.getScriptProperties();
  var r = UrlFetchApp.fetch('https://zoom.us/oauth/token?grant_type=account_credentials&account_id=' + encodeURIComponent(p.getProperty('ZOOM_ACCOUNT_ID')), {
    method: 'post', muteHttpExceptions: true,
    headers: { Authorization: 'Basic ' + Utilities.base64Encode(p.getProperty('ZOOM_CLIENT_ID') + ':' + p.getProperty('ZOOM_CLIENT_SECRET')) } });
  var j = JSON.parse(r.getContentText());
  if (!j.access_token) throw new Error('Zoom auth failed: ' + r.getContentText());
  c.put('zoom_tok', j.access_token, 3000);
  return j.access_token;
}

function zoomCalls_(fromDay, toDay) {
  var tok = zoomToken_(), out = [];
  var tries = [['https://api.zoom.us/v2/phone/call_history', 'call_logs'], ['https://api.zoom.us/v2/phone/call_logs', 'call_logs']];
  for (var i = 0; i < tries.length; i++) {
    var next = '', ok = true, got = [];
    do {
      var r = UrlFetchApp.fetch(tries[i][0] + '?from=' + fromDay + '&to=' + toDay + '&page_size=300' + (next ? '&next_page_token=' + encodeURIComponent(next) : ''),
        { headers: { Authorization: 'Bearer ' + tok }, muteHttpExceptions: true });
      if (r.getResponseCode() !== 200) { ok = false; console.warn(tries[i][0] + ' ' + r.getResponseCode() + ' ' + r.getContentText()); break; }
      var j = JSON.parse(r.getContentText());
      got = got.concat(j[tries[i][1]] || []); next = j.next_page_token || '';
    } while (next);
    if (ok) { out = got; break; }
  }
  return out;
}

function last10_(s) { var d = String(s || '').replace(/\D/g, ''); return d.length >= 10 ? d.slice(-10) : ''; }

function syncZoomCalls() {
  var tz = 'Europe/London', now = new Date();
  var calls = zoomCalls_(Utilities.formatDate(new Date(now.getTime() - 2 * 864e5), tz, 'yyyy-MM-dd'), Utilities.formatDate(now, tz, 'yyyy-MM-dd'));
  var props = PropertiesService.getScriptProperties();
  var seen = JSON.parse(props.getProperty('ZOOM_SEEN') || '[]');
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var tabs = [FB_TAB, WEB_TAB].map(function (n) {
    var sh = ss.getSheetByName(n); if (!sh) return null;
    var data = sh.getDataRange().getValues(), h = data[0].map(function (x) { return String(x).trim().toLowerCase().replace(/:$/, ''); });
    var cp = h.indexOf('phone number') > -1 ? h.indexOf('phone number') : h.indexOf('phone');
    return { sh: sh, data: data, cp: cp, cn: h.indexOf('notes'), cs: h.indexOf('status') };
  }).filter(function (t) { return t && t.cp > -1; });
  calls.sort(function (a, b) { return String(a.start_time || a.date_time).localeCompare(String(b.start_time || b.date_time)); });
  calls.forEach(function (c) {
    var id = String(c.id || c.call_id || ''); if (!id || seen.indexOf(id) > -1) return;
    var out = String(c.direction || '').toLowerCase() !== 'inbound';
    var other = out ? (c.callee_did_number || c.callee_number || c.callee_ext_number) : (c.caller_did_number || c.caller_number || c.caller_ext_number);
    var key = last10_(other); if (!key) return;
    var dur = +c.duration || 0, res = String(c.call_result || c.result || '').toLowerCase().replace(/_/g, ' ');
    var answered = /answer|connect|success/.test(res) && !/no answer|not answer|unanswer/.test(res) || (!res && dur > 0);
    var when = Utilities.formatDate(new Date(c.start_time || c.date_time), tz, 'dd/MM HH:mm');
    var mins = dur >= 60 ? Math.floor(dur / 60) + 'm ' + (dur % 60) + 's' : dur + 's';
    var txt = when + ' ' + (out ? 'called them' : 'they called us') + ' - ' + (answered ? 'answered (' + mins + ')' : (res || 'no answer'));
    var hit = false;
    tabs.forEach(function (t) {
      if (hit) return;
      for (var r = t.data.length - 1; r > 0; r--) {
        if (last10_(t.data[r][t.cp]) !== key) continue;
        hit = true;
        if (t.cn > -1) { var n = String(t.data[r][t.cn] || ''); n = (n ? n + ' | ' : '') + 'CALL: ' + txt; t.sh.getRange(r + 1, t.cn + 1).setValue(n); t.data[r][t.cn] = n; }
        if (t.cs > -1) {
          var st = String(t.data[r][t.cs] || '');
          if (!/call booked/i.test(st)) {
            var tries = (String(t.data[r][t.cn] || '').match(/CALL: [^|]*called them - (?!answered)/g) || []).length;
            st = answered ? 'Spoke on phone' : (out ? 'Tried calling (' + tries + ')' : 'Missed their call - call back');
            t.sh.getRange(r + 1, t.cs + 1).setValue(st); t.data[r][t.cs] = st;
          }
        }
        break;
      }
    });
    seen.push(id);
  });
  props.setProperty('ZOOM_SEEN', JSON.stringify(seen.slice(-800)));
}
