// Lera lead capture + CRM (runs as info@lera-realestate.com)
// Landing pages post each lead here; it lands as a new row on the "Leads" tab of the Google Sheet.
// The Sheet has four contact tabs. A contact's Status decides which tab they sit on, and the row moves by itself
// whenever the Status changes (typed by hand, or set by this script from calls, bookings and emails).

var SHEET_ID = '1_uUz0BLJT80qdkv2A_wfQEVqvDkeH4QXMyo2CBVooq4';
var TAB_LEADS = 'Leads', TAB_QUAL = 'Qualified contacts', TAB_ACTIVE = 'Active investors with us', TAB_DONE = 'Done a deal with us';
var CRM_TABS = [TAB_LEADS, TAB_QUAL, TAB_ACTIVE, TAB_DONE];
var CRM_HEADERS = ['Added', 'Name', 'Phone', 'Email', 'Source', 'Status', 'Budget', 'Looking for', 'Area', 'Deal / property',
  'Next step', 'Last contacted', 'Notes', 'Communication log', 'Campaign', 'Ad', 'Landing page', 'Meta click ID', 'Deal plan link'];
var CRM_WIDTHS = [90, 170, 125, 210, 150, 130, 150, 220, 170, 240, 240, 105, 320, 360, 170, 170, 100, 120, 120];
// Status -> tab. Anything not listed stays where it is.
var STATUS_TAB = {
  'new': TAB_LEADS, 'contacted': TAB_LEADS, 'tried calling': TAB_LEADS, 'nurture': TAB_LEADS, 'not interested': TAB_LEADS, 'invalid details': TAB_LEADS,
  'call booked': TAB_QUAL, 'spoke on phone': TAB_QUAL, 'qualified': TAB_QUAL, 'deal presented': TAB_QUAL, 'negotiating': TAB_QUAL, 'not proceeding': TAB_QUAL,
  'active deal': TAB_ACTIVE, 'deal done': TAB_DONE
};
var STATUS_LIST = ['New', 'Contacted', 'Tried calling', 'Nurture', 'Not interested', 'Invalid details', 'Call booked', 'Spoke on phone',
  'Qualified', 'Deal presented', 'Negotiating', 'Not proceeding', 'Active deal', 'Deal done'];
// How far along each status is. The script only ever moves someone forward, never back.
var STATUS_RANK = { 'new': 0, 'contacted': 1, 'tried calling': 1, 'nurture': 2, 'not interested': 2, 'invalid details': 2, 'call booked': 3,
  'spoke on phone': 4, 'qualified': 5, 'deal presented': 6, 'negotiating': 7, 'not proceeding': 7, 'active deal': 8, 'deal done': 9 };
var LEAD_KEY = 'lera-7Qm4xV9pT2'; // must match the key sent by the landing page
var ME = 'info@lera-realestate.com';

/* Run once: styles the four tabs, adds the Status dropdown, and starts the automatic jobs
   (row moves on Status edits, Gmail logging every 10 minutes). Safe to run again. */
function setupCrm() {
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var colours = ['#9a732f', '#2f6b4f', '#1f4e79', '#6b3fa0'];
  CRM_TABS.forEach(function (n, i) {
    var sh = ss.getSheetByName(n) || ss.insertSheet(n, i);
    if (!String(sh.getRange(1, 1).getValue())) sh.getRange(1, 1, 1, CRM_HEADERS.length).setValues([CRM_HEADERS]);
    styleTab_(sh, colours[i]);
  });
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (['onSheetEdit', 'syncGmail'].indexOf(t.getHandlerFunction()) > -1) ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger('onSheetEdit').forSpreadsheet(ss).onEdit().create();
  ScriptApp.newTrigger('syncGmail').timeBased().everyMinutes(10).create();
  var p = PropertiesService.getScriptProperties();
  if (!p.getProperty('GMAIL_LAST')) p.setProperty('GMAIL_LAST', String(Math.floor(Date.now() / 1000)));
  console.log('CRM ready: ' + CRM_TABS.map(function (n) { return n + ' (' + Math.max(ss.getSheetByName(n).getLastRow() - 1, 0) + ')'; }).join(', '));
}

function styleTab_(sh, colour) {
  var nc = CRM_HEADERS.length, nr = Math.max(sh.getMaxRows(), 200);
  if (sh.getMaxRows() < nr) sh.insertRowsAfter(sh.getMaxRows(), nr - sh.getMaxRows());
  if (sh.getMaxColumns() > nc) sh.deleteColumns(nc + 1, sh.getMaxColumns() - nc);
  sh.setFrozenRows(1); sh.setFrozenColumns(2); sh.setHiddenGridlines(true); sh.setTabColor(colour);
  sh.getBandings().forEach(function (b) { b.remove(); });
  sh.getRange(1, 1, nr, nc).applyRowBanding().setHeaderRowColor('#1a130d').setFirstRowColor('#ffffff').setSecondRowColor('#f5f5f3');
  sh.getRange(1, 1, 1, nc).setFontFamily('Jost').setFontSize(10).setFontWeight('bold').setFontColor('#faf5ed')
    .setHorizontalAlignment('center').setVerticalAlignment('middle').setWrap(true)
    .setBorder(null, null, true, null, null, null, '#9a732f', SpreadsheetApp.BorderStyle.SOLID_MEDIUM);
  sh.setRowHeight(1, 44);
  var body = sh.getRange(2, 1, nr - 1, nc);
  body.setFontFamily('Figtree').setFontSize(10).setFontColor('#1a130d').setFontWeight('normal').setVerticalAlignment('middle')
    .setWrapStrategy(SpreadsheetApp.WrapStrategy.CLIP).setNumberFormat('@')
    .setBorder(null, null, true, null, null, true, '#e7ddcc', SpreadsheetApp.BorderStyle.SOLID);
  sh.getRange(2, 2, nr - 1, 1).setFontWeight('bold');
  sh.getRange(2, 6, nr - 1, 1).setHorizontalAlignment('center');
  sh.setRowHeightsForced(2, nr - 1, 30);
  CRM_WIDTHS.forEach(function (w, i) { sh.setColumnWidth(i + 1, w); });
  sh.getRange(2, 6, nr - 1, 1).setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(STATUS_LIST, true).setAllowInvalid(true).build());
  var st = sh.getRange(2, 6, nr - 1, 1), rule = function (txt, bg, fg) {
    return SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo(txt).setBackground(bg).setFontColor(fg).setRanges([st]).build(); };
  sh.setConditionalFormatRules([
    rule('Active deal', '#dcefe3', '#1d5b3a'), rule('Deal done', '#e9e0f5', '#4b2c7a'), rule('Deal presented', '#f6ead2', '#7a5516'),
    rule('Negotiating', '#f6ead2', '#7a5516'), rule('Call booked', '#e3edf7', '#1f4e79'), rule('Spoke on phone', '#e3edf7', '#1f4e79'),
    rule('Qualified', '#e3edf7', '#1f4e79'), rule('Not interested', '#eeeeee', '#777777'), rule('Invalid details', '#eeeeee', '#777777'),
    rule('Not proceeding', '#eeeeee', '#777777')]);
  if (sh.getFilter()) sh.getFilter().remove();
  sh.getRange(1, 1, nr, nc).createFilter();
}

/* ---------- finding contacts and moving them between tabs ---------- */
function headers_(sh) {
  return sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(function (x) { return String(x).trim().toLowerCase().replace(/:$/, '').trim(); });
}
function last10_(s) { var d = String(s || '').replace(/\D/g, ''); return d.length >= 10 ? d.slice(-10) : ''; }
function today_() { return Utilities.formatDate(new Date(), 'Europe/London', 'dd/MM/yyyy'); }

// Finds a contact on any of the four tabs by email and/or phone. Returns {sh, row, h, v} or null.
function findContact_(email, phone) {
  var ss = SpreadsheetApp.openById(SHEET_ID), em = String(email || '').trim().toLowerCase(), ph = last10_(phone);
  if (!em && !ph) return null;
  for (var t = 0; t < CRM_TABS.length; t++) {
    var sh = ss.getSheetByName(CRM_TABS[t]); if (!sh || sh.getLastRow() < 2) continue;
    var data = sh.getDataRange().getValues(), h = data[0].map(function (x) { return String(x).trim().toLowerCase(); });
    var ce = h.indexOf('email'), cp = h.indexOf('phone');
    for (var r = data.length - 1; r > 0; r--) {
      if ((em && String(data[r][ce]).trim().toLowerCase() === em) || (ph && last10_(data[r][cp]) === ph))
        return { sh: sh, row: r + 1, h: h, v: data[r] };
    }
  }
  return null;
}

function setCell_(c, col, val) {
  var k = c.h.indexOf(col); if (k < 0) return;
  c.sh.getRange(c.row, k + 1).setValue(val); c.v[k] = val;
}
function getCell_(c, col) { var k = c.h.indexOf(col); return k < 0 ? '' : String(c.v[k] || ''); }

// Adds a line to the contact's Communication log and stamps Last contacted
function logComm_(c, line, contacted) {
  var log = getCell_(c, 'communication log');
  setCell_(c, 'communication log', (log ? log + ' | ' : '') + line);
  if (contacted !== false) setCell_(c, 'last contacted', today_());
}

// Moves a contact forward to `status` (never backwards), then puts the row on the right tab
function advance_(c, status) {
  var cur = getCell_(c, 'status').toLowerCase().replace(/\s*\(.*$/, '').trim();
  var a = STATUS_RANK.hasOwnProperty(cur) ? STATUS_RANK[cur] : 0, b = STATUS_RANK[String(status).toLowerCase()];
  if (b === undefined || b <= a) return c;
  setCell_(c, 'status', status);
  try { metaStage_(c.sh, c.row, status); } catch (x) { console.error(x); }
  return moveRow_(c.sh, c.row) || c;
}

// Puts a row on the tab its Status belongs to. Returns the contact at its new place (or null if it didn't move).
function moveRow_(sh, row) {
  var h = headers_(sh), v = sh.getRange(row, 1, 1, h.length).getValues()[0];
  var target = STATUS_TAB[String(v[h.indexOf('status')] || '').toLowerCase().replace(/\s*\(.*$/, '').trim()];
  if (!target || target === sh.getName() || !String(v[h.indexOf('name')] || v[h.indexOf('email')] || v[h.indexOf('phone')])) return null;
  var to = sh.getParent().getSheetByName(target); if (!to) return null;
  var th = headers_(to), out = th.map(function (x) { var k = h.indexOf(x); return k < 0 ? '' : v[k]; });
  var r = to.getLastRow() + 1;
  to.getRange(r, 1, 1, out.length).setNumberFormat('@').setValues([out]);
  sh.deleteRow(row);
  return { sh: to, row: r, h: th, v: out };
}

/* ---------- leads from the website ---------- */
function doPost(e) {
  var p = (e && e.parameter) || {};
  if (p._honey) return out_({ ok: true });
  if (p.key !== LEAD_KEY) return out_({ ok: false });
  if (p.action === 'book') return out_(book_(p));

  // Ad tags: fall back to the full page address if the page couldn't pass them separately
  if (!p.utm_source && p.url) {
    var qs = String(p.url).split('?')[1] || '';
    qs.split('#')[0].split('&').forEach(function (kv) {
      var i = kv.indexOf('='); if (i < 1) return;
      var k = decodeURIComponent(kv.slice(0, i)), v = decodeURIComponent(kv.slice(i + 1).replace(/\+/g, ' '));
      if (/^(utm_source|utm_campaign|utm_content|fbclid)$/.test(k) && !p[k]) p[k] = v;
    });
  }

  var src = String(p.utm_source || p.source || '').toLowerCase();
  var isAd = /facebook|instagram|^fb|^ig|meta/.test(src) || !!p.fbclid;
  var name = String(p.name || [p.first_name, p.last_name].filter(String).join(' ') || '').trim();
  var when = Utilities.formatDate(new Date(), 'Europe/London', 'dd/MM HH:mm');
  var action = p.action === 'call' ? 'Booked / requested a call' : p.action === 'enquiry' ? 'Website enquiry' : 'Downloaded HMO guide';
  var line = when + ' ' + action + (p.message ? ': ' + p.message : '') + (p.page ? ' (' + p.page + ')' : '');

  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var c = findContact_(p.email, p.phone);
    if (c) {                                    // already on the Sheet: add to their record instead of a duplicate row
      logComm_(c, line, false);
      if (!getCell_(c, 'budget') && p.budget) setCell_(c, 'budget', p.budget);
      if (!getCell_(c, 'phone') && p.phone) setCell_(c, 'phone', String(p.phone));
      if (!getCell_(c, 'meta click id') && p.fbclid) setCell_(c, 'meta click id', p.fbclid);
    } else {
      var sh = SpreadsheetApp.openById(SHEET_ID).getSheetByName(TAB_LEADS);
      var values = {
        'added': Utilities.formatDate(new Date(), 'Europe/London', 'dd/MM/yyyy'),
        'name': name, 'phone': String(p.phone || ''), 'email': String(p.email || ''),
        'source': isAd ? (src.indexOf('insta') > -1 ? 'Instagram ad' : 'Facebook ad') : 'Website' + (p.referrer ? ' (came from ' + p.referrer + ')' : ''),
        'status': 'New', 'budget': p.budget || '',
        'looking for': p.action === 'call' ? 'Asked for a call' : p.action === 'enquiry' ? 'Website enquiry' : 'Downloaded HMO guide',
        'notes': p.message || '', 'communication log': line, 'last contacted': '',
        'campaign': p.utm_campaign || '', 'ad': p.utm_content || '', 'landing page': p.page || '', 'meta click id': p.fbclid || ''
      };
      var h = headers_(sh), r = sh.getLastRow() + 1;
      sh.getRange(r, 1, 1, h.length).setNumberFormat('@').setValues([h.map(function (x) { return values.hasOwnProperty(x) ? values[x] : ''; })]);
    }
  } finally {
    lock.releaseLock();
  }
  try { notifyLead_(p); } catch (err) { console.error(err); }
  return out_({ ok: true });
}

function doGet(e) {
  var a = e && e.parameter && e.parameter.action;
  if (a === 'slots') return out_(slotsResult_());
  return out_({ ok: true, service: 'Lera lead capture' });
}

}

/* ---------- Gmail -> Sheet (every 10 minutes) ----------
   Every email to or from a contact on the Sheet is added to their Communication log, with Last contacted updated.
   Google Meet notes (Gemini) for a call are added to that contact's Notes, and the call moves them to "Spoke on phone".
   An email from us to a "New" lead moves them to "Contacted". */
function syncGmail() {
  var props = PropertiesService.getScriptProperties();
  var since = +(props.getProperty('GMAIL_LAST') || Math.floor(Date.now() / 1000) - 86400);
  var seen = JSON.parse(props.getProperty('GMAIL_SEEN') || '[]');
  var startedAt = Math.floor(Date.now() / 1000);
  var threads = GmailApp.search('after:' + (since - 120) + ' -in:spam -in:trash -from:submissions@formsubmit.co', 0, 100);
  var msgs = [];
  threads.forEach(function (t) { t.getMessages().forEach(function (m) { if (m.getDate().getTime() / 1000 >= since - 120 && seen.indexOf(m.getId()) < 0) msgs.push(m); }); });
  msgs.sort(function (a, b) { return a.getDate() - b.getDate(); });
  var addr = function (s) { return (String(s || '').match(/[\w.+'-]+@[\w-]+(\.[\w-]+)+/g) || []).map(function (x) { return x.toLowerCase(); }); };
  msgs.forEach(function (m) {
    seen.push(m.getId());
    var from = addr(m.getFrom())[0] || '', subj = m.getSubject() || '(no subject)', when = Utilities.formatDate(m.getDate(), 'Europe/London', 'dd/MM HH:mm');
    try {
      if (from === 'gemini-notes@google.com') return meetNotes_(m, subj, when);
      if (/^mailer-daemon|^postmaster/.test(from)) return;
      var out = from === ME, people = out ? addr(m.getTo() + ',' + m.getCc() + ',' + m.getBcc()) : [from];
      people = people.filter(function (x) { return x !== ME && !/@lera-realestate\.com$/.test(x); });
      if (out && people.length > 5) return;      // newsletters / group sends
      var snip = m.getPlainBody().replace(/\r/g, '').split(/\nOn .{5,80}wrote:|\n>|\nKind regards|\n-- ?\n/)[0].replace(/\s+/g, ' ').trim().slice(0, 140);
      people.forEach(function (who) {
        var c = findContact_(who); if (!c) return;
        logComm_(c, when + ' EMAIL ' + (out ? 'to them' : 'from them') + ': ' + subj + (snip ? ' - ' + snip : ''));
        if (out) advance_(c, 'Contacted');
      });
    } catch (x) { console.error('Gmail sync ' + m.getId() + ': ' + x); }
  });
  props.setProperty('GMAIL_SEEN', JSON.stringify(seen.slice(-1500)));
  props.setProperty('GMAIL_LAST', String(startedAt));
}

// "Notes: 'Lera Real Estate – video call with Aman' 18 Nov 2026" -> Aman's row
function meetNotes_(m, subj, when) {
  var who = (subj.match(/call with ([^'’(]+)/i) || [])[1]; if (!who) return;
  who = who.trim().toLowerCase();
  var c = findByName_(who); if (!c) return;
  var body = m.getPlainBody().replace(/\r/g, '');
  var notes = (body.split(/Quick Notes|Summary/)[1] || body).split(/Meeting records|Is the content of this email helpful/)[0]
    .replace(/\n{2,}/g, '\n').replace(/\n/g, ' ').replace(/\s+/g, ' ').trim().slice(0, 2500);
  var n = getCell_(c, 'notes');
  setCell_(c, 'notes', (n ? n + '\n\n' : '') + 'CALL NOTES ' + when + ' (Gemini): ' + notes);
  logComm_(c, when + ' VIDEO/CALL: notes added (Gemini)');
  advance_(c, 'Spoke on phone');
}

function findByName_(who) {
  var ss = SpreadsheetApp.openById(SHEET_ID), hits = [];
  CRM_TABS.forEach(function (n) {
    var sh = ss.getSheetByName(n); if (!sh || sh.getLastRow() < 2) return;
    var data = sh.getDataRange().getValues(), h = data[0].map(function (x) { return String(x).trim().toLowerCase(); }), cn = h.indexOf('name');
    for (var r = 1; r < data.length; r++) {
      var nm = String(data[r][cn] || '').toLowerCase().replace(/\(.*?\)/g, ' ').replace(/\s+/g, ' ').trim();
      if (nm && (nm === who || nm.indexOf(who + ' ') === 0 || who.indexOf(nm) === 0)) hits.push({ sh: sh, row: r + 1, h: h, v: data[r] });
    }
  });
  return hits.length === 1 ? hits[0] : null;   // only when it's clearly one person
}

var BOOK_URL  = 'https://lera-realestate.com/guide/?book=1';
var GUIDE_URL = 'https://lera-realestate.com/lera-hmo-guide.pdf';

var WA_TOKEN  = '';          // WhatsApp Cloud API access token (filled in once WhatsApp is set up)

// Runs straight after a lead is saved
function notifyLead_(p) {
  var first = cap_(String(p.name || '').trim().split(/\s+/)[0] || 'there');
  var email = String(p.email || '').trim();
  if (p.action === 'enquiry') return; // a personal reply follows; no automatic guide email
  if (p.action === 'call') {
    return; // they pick a time on the page; booking sends the confirmation
    if (email) sendMail_(email, 'Your call with Lera: pick a time',
      'Hi ' + first + ',\n\nThanks for asking for a call. If you haven\'t picked a time yet, choose one here:\n' + BOOK_URL +
      '\n\nIt\'s 15 minutes with an HMO specialist. We\'ll go through what your budget could do and walk you through a real deal line by line.\n\nLera Real Estate\n' + REPLY_TO);
    sendWa_(p.phone, 'lera_call_request', [first]);
  } else {
    if (email) sendMail_(email, 'Your free HMO guide is here, ' + first,
      'Hi ' + first + ',\n\nHere\'s your copy of the Lera HMO guide: ' + GUIDE_URL + '\n\nWant to talk it through? Book a free call with Akiva: ' + BOOK_URL + '\n\nLera Real Estate', guideEmail_(first));
    sendWa_(p.phone, 'lera_guide', [first]);
  }
}

// Runs every minute: reminds people 1 hour and 5 minutes before their call
function callReminders() {
  var cal = CalendarApp.getCalendarById(CAL_ID) || CalendarApp.getDefaultCalendar();
  var me = Session.getEffectiveUser().getEmail().toLowerCase();
  var now = new Date();
  var events = cal.getEvents(now, new Date(now.getTime() + 66 * 60000));
  events.forEach(function (ev) {
    var guests = ev.getGuestList().filter(function (g) { return g.getEmail().toLowerCase() !== me; });
    if (!guests.length) return;                       // not a booked call
    var g = guests[0], email = g.getEmail();
    var mins = (ev.getStartTime().getTime() - now.getTime()) / 60000;
    var lead = findLead_(email);
    var first = cap_((lead.first || g.getName() || '').split(/\s+/)[0] || 'there');
    var phone = lead.phone || phoneFrom_(ev.getDescription());
    var time = Utilities.formatDate(ev.getStartTime(), 'Europe/London', 'h:mm a');
    if (!/^(Free (video |phone )?call with Lera|Lera Real Estate [–-] (video|phone) call with)/.test(ev.getTitle())) return; // only Lera calls
    var link = meetLink_(ev);
    if (mins <= 62 && mins > 50 && !ev.getTag('lera_r60')) {
      sendMail_(email, 'In one hour: your call with Akiva at ' + time,
        'Hi ' + first + ', your call with Akiva is in one hour, at ' + time + '.' + (link ? ' Join: ' + link : ' Akiva will call you on ' + phone + '.'), reminderEmail_(first, time, !!link, link, phone, false));
      ev.setTag('lera_r60', '1');
    }
    if (mins <= 11 && mins > 0 && !ev.getTag('lera_r5')) {
      sendMail_(email, 'Starting now: your call with Akiva',
        'Hi ' + first + ', your call with Akiva starts in 10 minutes, at ' + time + '.' + (link ? ' Join: ' + link : ' Akiva will call you on ' + phone + '.'), reminderEmail_(first, time, !!link, link, phone, true));
      sendWa_(phone, 'lera_call_reminder_5min', [first, time]);
      ev.setTag('lera_r5', '1');
    }
  });
}

function findLead_(email) {
  var c = findContact_(email);
  return c ? { phone: getCell_(c, 'phone').replace(/^'/, ''), first: getCell_(c, 'name').replace(/\(.*?\)/g, '').trim() } : {};
}

function phoneFrom_(text) { var m = String(text || '').match(/(\+?44|0)\s?7[\d\s]{9,12}/); return m ? m[0] : ''; }
function meetLink_(ev) { var m = String(ev.getDescription() || '').match(/https:\/\/meet\.google\.com\/[a-z\-]+/); return m ? m[0] : ''; }

function sendMail_(to, subject, body, html) {
  // Only ever sends from info@lera-realestate.com
  try {
    if (Session.getEffectiveUser().getEmail().toLowerCase() !== REPLY_TO) { console.error('Not running as info@ - email NOT sent to ' + to); return; }
    var o = { name: FROM_NAME, replyTo: REPLY_TO }; if (html) o.htmlBody = html;
    GmailApp.sendEmail(to, subject, body, o);
  }
  catch (e) { console.error('email failed ' + to + ': ' + e); }
}

// WhatsApp needs approved message templates with these names (body variables {{1}}, {{2}}).
function sendWa_(phone, template, vars) {
  var props = PropertiesService.getScriptProperties();
  var tok = props.getProperty('WA_TOKEN') || WA_TOKEN, pid = props.getProperty('WA_PHONE_ID') || WA_PHONE_ID;
  if (!tok || !pid || !phone) return;
  var n = String(phone).replace(/[^\d+]/g, '');
  if (/^07\d{9}$/.test(n)) n = '44' + n.slice(1);
  n = n.replace(/^\+/, '').replace(/^0044/, '44');
  var payload = { messaging_product: 'whatsapp', to: n, type: 'template',
    template: { name: template, language: { code: 'en_GB' },
      components: [{ type: 'body', parameters: vars.map(function (v) { return { type: 'text', text: String(v) }; }) }] } };
  try {
    var r = UrlFetchApp.fetch('https://graph.facebook.com/v21.0/' + pid + '/messages', {
      method: 'post', contentType: 'application/json', muteHttpExceptions: true,
      headers: { Authorization: 'Bearer ' + tok }, payload: JSON.stringify(payload) });
    if (r.getResponseCode() >= 300) console.error('WhatsApp ' + template + ' failed: ' + r.getContentText());
  } catch (e) { console.error('WhatsApp failed: ' + e); }
}

// Run once: starts the reminder clock
function setup() {
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'callReminders') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('callReminders').timeBased().everyMinutes(1).create();

// ===================== BOOKING (calls booked straight from the landing page) =====================
var SLOT_DAYS = 10;           // working days shown
// England & Wales bank holidays (gov.uk). No calls are offered on these days, or at weekends.
var BANK_HOLIDAYS = ['2026-01-01','2026-04-03','2026-04-06','2026-05-04','2026-05-25','2026-08-31','2026-12-25','2026-12-28',
  '2027-01-01','2027-03-26','2027-03-29','2027-05-03','2027-05-31','2027-08-30','2027-12-27','2027-12-28',
  '2028-01-03','2028-04-14','2028-04-17','2028-05-01','2028-05-29','2028-08-28','2028-12-25','2028-12-26'];
function closedDay_(d) { var tz = 'Europe/London'; return +Utilities.formatDate(d, tz, 'u') > 5 || BANK_HOLIDAYS.indexOf(Utilities.formatDate(d, tz, 'yyyy-MM-dd')) > -1; }
var SLOT_FIRST_HOUR = 9;      // first call 9:00
var SLOT_LAST_START = '16:30';// last call 16:30
var CALL_MINS = 15;           // length of the call
var MIN_NOTICE_HOURS = 2;     // earliest bookable slot from now
var OLD_SCHEDULE_TITLE = 'Free 15-minute HMO strategy call';

function cal_() { return CalendarApp.getCalendarById(CAL_ID) || CalendarApp.getDefaultCalendar(); }
function pad_(n) { return (n < 10 ? '0' : '') + n; }

function blocks_(ev) {
  if (ev.isAllDayEvent()) return false;
  try { if (ev.getTransparency && String(ev.getTransparency()) === 'TRANSPARENT') return false; } catch (x) {}
  if (ev.getTitle() === OLD_SCHEDULE_TITLE && ev.getGuestList().length === 0) return false; // the old availability block
  return true;
}

function slotsResult_() {
  var tz = 'Europe/London', now = new Date(), minStart = now.getTime() + MIN_NOTICE_HOURS * 3600000;
  var horizon = new Date(now.getTime() + 21 * 864e5);
  var evs = cal_().getEvents(now, horizon).filter(blocks_);
  var lastH = +SLOT_LAST_START.split(':')[0], lastM = +SLOT_LAST_START.split(':')[1];
  var days = [];
  for (var i = 0; i < 21 && days.length < SLOT_DAYS; i++) {
    var day = new Date(now.getTime() + i * 864e5);
    if (closedDay_(day)) continue; // weekdays only, no bank holidays
    var ymd = Utilities.formatDate(day, tz, 'yyyy-MM-dd'), times = [];
    for (var h = SLOT_FIRST_HOUR; h <= lastH; h++) {
      for (var m = 0; m < 60; m += 30) {
        if (h === lastH && m > lastM) continue;
        var s = Utilities.parseDate(ymd + ' ' + pad_(h) + ':' + pad_(m), tz, 'yyyy-MM-dd HH:mm').getTime();
        var e = s + CALL_MINS * 60000;
        if (s < minStart) continue;
        var clash = evs.some(function (ev) { return ev.getStartTime().getTime() < e && ev.getEndTime().getTime() > s; });
        if (!clash) times.push(new Date(s).toISOString());
      }
    }
    if (times.length) days.push({ date: ymd, label: Utilities.formatDate(day, tz, 'EEE d MMM'), times: times });
  }
  return { ok: true, days: days };
}

/* ===================== Zoom Phone call logging =====================
   Every 15 min: reads Zoom Phone call history and notes each call to/from a lead in the Sheet.
   Script properties needed (Project Settings > Script properties): ZOOM_ACCOUNT_ID, ZOOM_CLIENT_ID, ZOOM_CLIENT_SECRET
   Then run setupZoom() once. */
function setupZoom() {
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'syncZoomCalls') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('syncZoomCalls').timeBased().everyMinutes(15).create();
  syncZoomCalls();
}

function zoomToken_() {
  var c = CacheService.getScriptCache(), t = c.get('zoom_tok'); if (t) return t;
  var p = PropertiesService.getScriptProperties();
  var r = UrlFetchApp.fetch('https://zoom.us/oauth/token?grant_type=account_credentials&account_id=' + encodeURIComponent(p.getProperty('ZOOM_ACCOUNT_ID')), {
    method: 'post', muteHttpExceptions: true,
    headers: { Authorization: 'Basic ' + Utilities.base64Encode(p.getProperty('ZOOM_CLIENT_ID') + ':' + p.getProperty('ZOOM_CLIENT_SECRET')) } });
  var j = JSON.parse(r.getContentText());
  if (!j.access_token) throw new Error('Zoom auth failed: ' + r.getContentText());
  c.put('zoom_tok', j.access_token, 3000);
  return j.access_token;
}

function zoomCalls_(fromDay, toDay) {
  var tok = zoomToken_(), out = [];
  var tries = [['https://api.zoom.us/v2/phone/call_history', 'call_logs'], ['https://api.zoom.us/v2/phone/call_logs', 'call_logs']];
  for (var i = 0; i < tries.length; i++) {
    var next = '', ok = true, got = [];
    do {
      var r = UrlFetchApp.fetch(tries[i][0] + '?from=' + fromDay + '&to=' + toDay + '&page_size=300' + (next ? '&next_page_token=' + encodeURIComponent(next) : ''),
        { headers: { Authorization: 'Bearer ' + tok }, muteHttpExceptions: true });
      if (r.getResponseCode() !== 200) { ok = false; console.warn(tries[i][0] + ' ' + r.getResponseCode() + ' ' + r.getContentText()); break; }
      var j = JSON.parse(r.getContentText());
      got = got.concat(j[tries[i][1]] || []); next = j.next_page_token || '';
    } while (next);
    if (ok) { out = got; break; }
  }
  return out;
}

function syncZoomCalls() {
  var tz = 'Europe/London', now = new Date();
  var calls = zoomCalls_(Utilities.formatDate(new Date(now.getTime() - 2 * 864e5), tz, 'yyyy-MM-dd'), Utilities.formatDate(now, tz, 'yyyy-MM-dd'));
  var props = PropertiesService.getScriptProperties();
  var seen = JSON.parse(props.getProperty('ZOOM_SEEN') || '[]');
  calls.sort(function (a, b) { return String(a.start_time || a.date_time).localeCompare(String(b.start_time || b.date_time)); });
  calls.forEach(function (c) {
    var id = String(c.id || c.call_id || ''); if (!id || seen.indexOf(id) > -1) return;
    seen.push(id);
    var out = String(c.direction || '').toLowerCase() !== 'inbound';
    var other = out ? (c.callee_did_number || c.callee_number || c.callee_ext_number) : (c.caller_did_number || c.caller_number || c.caller_ext_number);
    if (!last10_(other)) return;
    var dur = +c.duration || 0, res = String(c.call_result || c.result || '').toLowerCase().replace(/_/g, ' ');
    var answered = /answer|connect|success/.test(res) && !/no answer|not answer|unanswer/.test(res) || (!res && dur > 0);
    var when = Utilities.formatDate(new Date(c.start_time || c.date_time), tz, 'dd/MM HH:mm');
    var mins = dur >= 60 ? Math.floor(dur / 60) + 'm ' + (dur % 60) + 's' : dur + 's';
    var k = findContact_('', other); if (!k) return;
    logComm_(k, when + ' CALL ' + (out ? 'we called' : 'they called') + ': ' + (answered ? 'answered (' + mins + ')' : (res || 'no answer')));
    if (answered) advance_(k, 'Spoke on phone');
    else if (out) advance_(k, 'Tried calling');
    else setCell_(k, 'next step', 'Call them back (missed their call ' + when + ')');
  });
  props.setProperty('ZOOM_SEEN', JSON.stringify(seen.slice(-800)));
}

function metaEvent_(status) {
  var s = String(status || '').toLowerCase();
  if (/not interested|not proceeding|invalid|unqualified|not qualified/.test(s)) return '';
  if (/call booked/.test(s)) return 'CallBooked';
  if (/spoke/.test(s)) return 'SpokeOnPhone';
  if (/qualified|deal presented|negotiating/.test(s)) return 'QualifiedLead';
  if (/active deal|deal done|client|won/.test(s)) return 'Client';
  return '';
}

function sha_(v) {
  v = String(v || '').trim().toLowerCase(); if (!v) return '';
  return Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, v, Utilities.Charset.UTF_8)
    .map(function (b) { return ('0' + (b & 255).toString(16)).slice(-2); }).join('');
}

  var g = function () { for (var i = 0; i < arguments.length; i++) { var k = h.indexOf(arguments[i]); if (k > -1 && v[k] !== '') return String(v[k]); } return ''; };
  var email = g('email', 'email address').trim().toLowerCase();
  var phone = g('phone number', 'phone').replace(/\D/g, '');
  if (/^0/.test(phone)) phone = '44' + phone.slice(1);
  if (!email && !phone) return;
  if (/@lera-realestate\.com$|^asolomon1703@/.test(email) || /^test\b/i.test(g('first name', 'name'))) return; // skip our own test rows
  var key = 'meta:' + sha_(email || phone) + ':' + ev;
  if (props.getProperty(key)) return; // already sent for this lead
  var first = g('first name') || g('name').split(/\s+/)[0] || '';
  var last = g('last name') || g('name').split(/\s+/).slice(1).join(' ');
  var ud = {};
  if (email) ud.em = [sha_(email)];
  if (phone) ud.ph = [sha_(phone)];
  if (first) ud.fn = [sha_(first)];
  if (last) ud.ln = [sha_(last)];
  var fbclid = g('meta click id'); if (fbclid) ud.fbc = 'fb.1.' + Date.now() + '.' + fbclid;
  var body = { data: [{
    event_name: ev, event_time: Math.floor(Date.now() / 1000), action_source: 'system_generated',
    event_id: key.slice(5), user_data: ud,
    custom_data: { event_source: 'crm', lead_event_source: 'Google Sheets', lead_status: String(status), campaign: g('campaign'), ad: g('ad') }
  }] };
  var test = props.getProperty('META_TEST_CODE'); if (test) body.test_event_code = test;
  var r = UrlFetchApp.fetch('https://graph.facebook.com/v21.0/' + META_PIXEL + '/events?access_token=' + encodeURIComponent(token),
    { method: 'post', contentType: 'application/json', payload: JSON.stringify(body), muteHttpExceptions: true });
  if (r.getResponseCode() === 200) props.setProperty(key, new Date().toISOString());
  else console.error('Meta ' + r.getResponseCode() + ' ' + r.getContentText());
}

// Run once after adding the token: sends the stage for every lead already past "New"
function metaBackfill() {
  var ss = SpreadsheetApp.openById(SHEET_ID);
  CRM_TABS.forEach(function (n) {
    var sh = ss.getSheetByName(n); if (!sh || sh.getLastRow() < 2) return;
    var cs = headers_(sh).indexOf('status'); if (cs < 0) return;
    var vals = sh.getRange(2, cs + 1, sh.getLastRow() - 1, 1).getValues();
    vals.forEach(function (x, i) { if (metaEvent_(x[0])) metaStage_(sh, i + 2, String(x[0])); });
  });
}
